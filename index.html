import javax.swing.*;
import java.awt.*;
import java.awt.event.*;
public class JavaEvent extends JFrame implements ActionListener{
    //Create the GUI components
    private JButton jb1 = new JButton(" 1 ");
    private JButton jb2 = new JButton(" 2 ");
    private JButton jb3 = new JButton(" 3 ");
    private JTextField jtDisplay = new JTextField("");
    /** Creates a new instance of javaEvent */
    public JavaEvent() {
        JPanel p1 = new JPanel(new GridLayout(1,1));
        p1.add(jtDisplay);
        
        JPanel p2 = new JPanel();
        p2.setLayout(new GridLayout(1,3));
        p2.add(jb1);p2.add(jb2);p2.add(jb3);
        
        setLayout(new BorderLayout());
        add(p1,BorderLayout.NORTH);
        add(p2,BorderLayout.CENTER);
        
        jb1.addActionListener(this);
        jb2.addActionListener(this);
        jb3.addActionListener(this);
    }
     public static void main(String[] args) 
    {
        //Create and set up the window.
        JavaEvent f = new JavaEvent();
	  f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);    
        //Display the window.
        f.setTitle("Java Event");
        f.setSize(200,100);
        f.setVisible(true);
    }   
     public void actionPerformed(ActionEvent e)
     {
         if(e.getSource() == jb1)
             jtDisplay.setText("You have pressed 1");
         else if(e.getSource() == jb2)
             jtDisplay.setText("You have pressed 2");
         else if(e.getSource() == jb3)
             jtDisplay.setText("You have pressed 3");
     }
